plan_updated: plan: {goal?, phases: [{name, status?: pending|active|done|blocked|dropped, tasks: [{id, title, status?: pending|active|done|blocked|dropped, route?: {agent?, model?, effort?}, verify?: {cmd, cwd?, timeout_ms?}}]}]}
phase_status_changed: phase, status: pending|active|done|blocked|dropped, note? (say why when dropped)
task_added: phase, id, title, status?: pending|active|done|blocked|dropped, verify?: {cmd, cwd?, timeout_ms?} (the acceptance command sofar drive runs before accepting the task)
task_status_changed: id, status: pending|active|done|blocked|dropped, note? (say why when blocked or dropped)
decision_logged: chose, over, because, rule? (one imperative every later session must obey), guard? (path:<globs> or cmd:<globs>; only with rule)
session_started: tool (your agent name), model?
session_ended: summary, next_action (the single next step)
session_closed: reason
file_touched: path, op
command_run: cmd
note_added: text
memory_promoted: text, supersedes? (qualified handle `<slug> M<n>` of the fact this one replaces)
review_recorded: scope: phase|final, verdict: pass|findings|blocked, watermark? (sha read through), phase?, findings?: string[] (required, non-empty, for findings)
run_started: run, adapter, policy: task|threshold, threshold_pct? and context_window? (both required for threshold), max_sessions?, surface?, verify? (the run's default acceptance command)
handoff: run, session_id, reason: task_done|threshold|stall|needs_user|verify_failed, task?, tokens?, detail? (how the process ended: exit, spawn error, last stderr line)
run_stopped: run, reason: closed|needs_user|stall|cost_cap|max_sessions|interrupted|error, note? (required for error)
run_stop_requested: run
verification_recorded: run, task, attempt, command, cwd, checked: {head, tree}, validator, result: pass|fail|timeout|error|refused, exit_code?, signal?, duration_ms, timeout_ms, diagnostics? (≤1,024 chars)
correction: ref (the bad event id), reason?
